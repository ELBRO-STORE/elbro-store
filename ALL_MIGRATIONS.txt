-- ELBRO STORE - Initial Schema
-- Categories, Products, Product Images, Product Variants

-- Enable required extensions
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
CREATE EXTENSION IF NOT EXISTS "pgcrypto";

-- Categories table
CREATE TABLE categories (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name TEXT NOT NULL,
  slug TEXT NOT NULL UNIQUE,
  description TEXT,
  image_url TEXT,
  display_order INTEGER DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Products table
CREATE TABLE products (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  category_id UUID NOT NULL REFERENCES categories(id) ON DELETE RESTRICT,
  name TEXT NOT NULL,
  slug TEXT NOT NULL UNIQUE,
  description TEXT NOT NULL,
  full_description TEXT,
  material TEXT,
  price_cents INTEGER NOT NULL CHECK (price_cents >= 0),
  sale_price_cents INTEGER CHECK (sale_price_cents >= 0),
  sku TEXT NOT NULL UNIQUE,
  featured BOOLEAN DEFAULT FALSE,
  best_seller BOOLEAN DEFAULT FALSE,
  published BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Product Images table
CREATE TABLE product_images (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  product_id UUID NOT NULL REFERENCES products(id) ON DELETE CASCADE,
  url TEXT NOT NULL,
  alt_text TEXT,
  is_main BOOLEAN DEFAULT FALSE,
  display_order INTEGER DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Product Variants table (size/color combinations with stock)
CREATE TABLE product_variants (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  product_id UUID NOT NULL REFERENCES products(id) ON DELETE CASCADE,
  size TEXT NOT NULL,
  color TEXT NOT NULL,
  stock INTEGER DEFAULT 0 CHECK (stock >= 0),
  sku TEXT NOT NULL UNIQUE,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(product_id, size, color)
);

-- Indexes for performance
CREATE INDEX idx_products_category_id ON products(category_id);
CREATE INDEX idx_products_published ON products(published) WHERE published = TRUE;
CREATE INDEX idx_products_featured ON products(featured) WHERE featured = TRUE;
CREATE INDEX idx_products_best_seller ON products(best_seller) WHERE best_seller = TRUE;
CREATE INDEX idx_products_slug ON products(slug);
CREATE INDEX idx_product_images_product_id ON product_images(product_id);
CREATE INDEX idx_product_variants_product_id ON product_variants(product_id);
CREATE INDEX idx_product_variants_stock ON product_variants(stock) WHERE stock > 0;

-- Updated_at trigger function
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Apply updated_at triggers
CREATE TRIGGER update_categories_updated_at
  BEFORE UPDATE ON categories
  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_products_updated_at
  BEFORE UPDATE ON products
  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_product_variants_updated_at
  BEFORE UPDATE ON product_variants
  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

-- Orders table
CREATE TABLE orders (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  order_ref TEXT NOT NULL UNIQUE,
  customer_name TEXT NOT NULL,
  customer_phone TEXT NOT NULL,
  customer_city TEXT NOT NULL,
  customer_address TEXT NOT NULL,
  customer_email TEXT,
  delivery_instructions TEXT,
  notes TEXT,
  subtotal_cents INTEGER NOT NULL CHECK (subtotal_cents >= 0),
  delivery_fee_cents INTEGER NOT NULL DEFAULT 0 CHECK (delivery_fee_cents >= 0),
  total_cents INTEGER NOT NULL CHECK (total_cents >= 0),
  status TEXT NOT NULL DEFAULT 'NEW' CHECK (status IN ('NEW', 'CONFIRMED', 'PREPARING', 'SHIPPED', 'DELIVERED', 'CANCELLED')),

  google_sheets_order_id TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()

  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

  product_id UUID NOT NULL REFERENCES products(id) ON DELETE RESTRICT,

  piece_name TEXT NOT NULL,
  size TEXT NOT NULL,
  color TEXT NOT NULL,
  quantity INTEGER NOT NULL CHECK (quantity > 0),
  unit_price_cents INTEGER NOT NULL CHECK (unit_price_cents >= 0),

  created_at TIMESTAMPTZ DEFAULT NOW()

CREATE INDEX idx_order_items_product_id ON order_items(product_id);

-- Trigger for updated_at

  BEFORE UPDATE ON orders
  FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

-- Function to decrement stock when order is created
CREATE OR REPLACE FUNCTION decrement_variant_stock()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.variant_id IS NOT NULL THEN
    UPDATE product_variants
    SET stock = stock - NEW.quantity
    WHERE id = NEW.variant_id
    AND stock >= NEW.quantity;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Insufficient stock for variant %', NEW.variant_id;
    END IF;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_decrement_stock
  AFTER INSERT ON order_items
  FOR EACH ROW EXECUTE FUNCTION decrement_variant_stock();

  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  role TEXT NOT NULL CHECK (role IN ('super_admin', 'admin', 'manager')) DEFAULT 'manager',
  created_at TIMESTAMPTZ DEFAULT NOW(),

  USING (auth.uid() = id);

  ON admin_users FOR ALL
  USING (

    )
  );

-- ELBRO STORE - Database Functions
-- Helper functions and triggers

-- Function to get store settings as a key-value map
CREATE OR REPLACE FUNCTION get_store_settings()
RETURNS JSONB AS $$
DECLARE
  settings JSONB := '{}'::JSONB;
  rec RECORD;
BEGIN
  FOR rec IN SELECT key, value FROM store_settings LOOP
    settings := jsonb_set(settings, ARRAY[rec.key], COALESCE(rec.value, 'null'::JSONB));
  END LOOP;
  RETURN settings;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to create order with items (atomic transaction)
CREATE OR REPLACE FUNCTION create_order(
  p_customer_name TEXT,
  p_customer_phone TEXT,
  p_customer_city TEXT,
  p_customer_address TEXT,
  p_customer_email TEXT,
  p_delivery_instructions TEXT,
  p_notes TEXT,
  p_items JSONB, -- Array of {product_id, variant_id, piece_name, size, color, quantity, unit_price_cents}
  p_delivery_fee_cents INTEGER DEFAULT 0
)
RETURNS UUID AS $$
DECLARE
  v_order_id UUID;
  v_order_ref TEXT;
  v_subtotal_cents INTEGER := 0;
  v_total_cents INTEGER;
  v_item JSONB;
BEGIN
  -- Generate order reference
  v_order_ref := 'ELBRO-' || TO_CHAR(NOW(), 'YYYYMMDD-HH24MISS') || '-' || UPPER(SUBSTRING(MD5(RANDOM()::TEXT) FROM 1 FOR 6));

  -- Calculate subtotal from items
  FOR v_item IN SELECT * FROM JSONB_ARRAY_ELEMENTS(p_items) LOOP
    v_subtotal_cents := v_subtotal_cents + (v_item->>'quantity')::INTEGER * (v_item->>'unit_price_cents')::INTEGER;
  END LOOP;

  v_total_cents := v_subtotal_cents + p_delivery_fee_cents;

  -- Create order
  INSERT INTO orders (
    order_ref,
    customer_name,
    customer_phone,
    customer_city,
    customer_address,
    customer_email,
    delivery_instructions,
    notes,
    subtotal_cents,
    delivery_fee_cents,
    total_cents,
    status
  ) VALUES (
    v_order_ref,
    p_customer_name,
    p_customer_phone,
    p_customer_city,
    p_customer_address,
    p_customer_email,
    p_delivery_instructions,
    p_notes,
    v_subtotal_cents,
    p_delivery_fee_cents,
    v_total_cents,
    'NEW'
  ) RETURNING id INTO v_order_id;

  -- Create order items
  FOR v_item IN SELECT * FROM JSONB_ARRAY_ELEMENTS(p_items) LOOP
    INSERT INTO order_items (
      order_id,
      product_id,
      variant_id,
      piece_name,
      size,
      color,
      quantity,
      unit_price_cents,
      subtotal_cents
    ) VALUES (
      v_order_id,
      (v_item->>'product_id')::UUID,
      CASE WHEN (v_item->>'variant_id') IS NOT NULL THEN (v_item->>'variant_id')::UUID ELSE NULL END,
      v_item->>'piece_name',
      v_item->>'size',
      v_item->>'color',
      (v_item->>'quantity')::INTEGER,
      (v_item->>'unit_price_cents')::INTEGER,
      (v_item->>'quantity')::INTEGER * (v_item->>'unit_price_cents')::INTEGER
    );
  END LOOP;

  RETURN v_order_id;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to update order status with validation
CREATE OR REPLACE FUNCTION update_order_status(
  p_order_id UUID,
  p_new_status TEXT,
  p_admin_user_id UUID
)
RETURNS BOOLEAN AS $$
DECLARE
  v_current_status TEXT;
  v_is_admin BOOLEAN;
BEGIN
  -- Check admin permission
  SELECT EXISTS(SELECT 1 FROM admin_users WHERE user_id = p_admin_user_id) INTO v_is_admin;
  IF NOT v_is_admin THEN
    RAISE EXCEPTION 'Unauthorized: Admin access required';
  END IF;

  -- Get current status
  SELECT status INTO v_current_status FROM orders WHERE id = p_order_id;
  IF v_current_status IS NULL THEN
    RAISE EXCEPTION 'Order not found';
  END IF;

  -- Validate status transition
  IF v_current_status = 'CANCELLED' AND p_new_status != 'CANCELLED' THEN
    RAISE EXCEPTION 'Cannot change status from CANCELLED';
  END IF;

  IF v_current_status = 'DELIVERED' AND p_new_status != 'DELIVERED' THEN
    RAISE EXCEPTION 'Cannot change status from DELIVERED';
  END IF;

  -- Update status
  UPDATE orders
  SET status = p_new_status, updated_at = NOW()
  WHERE id = p_order_id;

  RETURN TRUE;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to get product with variants and images (for product detail)
CREATE OR REPLACE FUNCTION get_product_detail(p_slug TEXT)
RETURNS JSONB AS $$
DECLARE
  v_result JSONB;
BEGIN
  SELECT jsonb_build_object(
    'product', to_jsonb(p),
    'images', (
      SELECT COALESCE(JSONB_AGG(to_jsonb(i) ORDER BY i.display_order), '[]'::JSONB)
      FROM product_images i
      WHERE i.product_id = p.id
    ),
    'variants', (
      SELECT COALESCE(JSONB_AGG(to_jsonb(v) ORDER BY v.size, v.color), '[]'::JSONB)
      FROM product_variants v
      WHERE v.product_id = p.id
    ),
    'category', to_jsonb(c)
  ) INTO v_result
  FROM products p
  JOIN categories c ON c.id = p.category_id
  WHERE p.slug = p_slug AND p.published = TRUE;

  RETURN v_result;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to get products with filters (for listing)
CREATE OR REPLACE FUNCTION get_products_filtered(
  p_category_slug TEXT DEFAULT NULL,
  p_search TEXT DEFAULT NULL,
  p_featured BOOLEAN DEFAULT NULL,
  p_best_seller BOOLEAN DEFAULT NULL,
  p_sort TEXT DEFAULT 'newest', -- newest, price_asc, price_desc, best_seller
  p_limit INTEGER DEFAULT 20,
  p_offset INTEGER DEFAULT 0
)
RETURNS JSONB AS $$
DECLARE
  v_query TEXT;
  v_count INTEGER;
  v_results JSONB;
BEGIN
  -- Build dynamic query
  v_query := '
    SELECT jsonb_build_object(
      ''products'', COALESCE(JSONB_AGG(to_jsonb(p) ORDER BY
        CASE
          WHEN $6 = ''price_asc'' THEN p.price_cents
          WHEN $6 = ''price_desc'' THEN p.price_cents DESC
          WHEN $6 = ''best_seller'' THEN p.best_seller DESC
          ELSE p.created_at DESC
        END
      ), ''[]''::JSONB),
      ''total'', COUNT(*) OVER()
    )
    FROM products p
    JOIN categories c ON c.id = p.category_id
    WHERE p.published = TRUE
  ';

  IF p_category_slug IS NOT NULL THEN
    v_query := v_query || ' AND c.slug = $1';
  END IF;

  IF p_search IS NOT NULL THEN
    v_query := v_query || ' AND (p.name ILIKE $2 OR p.description ILIKE $2)';
  END IF;

  IF p_featured IS NOT NULL THEN
    v_query := v_query || ' AND p.featured = $3';
  END IF;

  IF p_best_seller IS NOT NULL THEN
    v_query := v_query || ' AND p.best_seller = $4';
  END IF;

  v_query := v_query || ' LIMIT $5 OFFSET $6';

  -- This is a simplified version; in production use proper dynamic SQL
  RETURN '{}'::JSONB;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Safely links guest orders to authenticated user by verifying BOTH phone AND email
-- Only links orders where: user_id IS NULL AND customer_phone matches profile phone
-- AND (customer_email IS NULL OR customer_email matches profile email)
-- This prevents phone collision attacks where multiple users share a phone number.

CREATE OR REPLACE FUNCTION link_guest_orders_to_user(p_user_id UUID, p_phone TEXT)
RETURNS INTEGER AS $$
DECLARE
  v_updated INTEGER := 0;
  v_email TEXT;
BEGIN
  -- Get user's email from auth.users
  SELECT email INTO v_email FROM auth.users WHERE id = p_user_id;

  -- Only link orders matching BOTH phone AND email (when email exists)
  UPDATE orders
  SET user_id = p_user_id
  WHERE user_id IS NULL
    AND customer_phone = p_phone
    AND (v_email IS NULL OR customer_email = v_email);

  GET DIAGNOSTICS v_updated = ROW_COUNT;
  RETURN v_updated;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- ============================================================

CREATE TABLE coupons (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  code TEXT NOT NULL UNIQUE, -- e.g., "WELCOME10", "SUMMER20"
  name TEXT NOT NULL, -- e.g., "Welcome Discount", "Summer Sale"
  description TEXT,

  -- Discount configuration
  discount_type TEXT NOT NULL CHECK (discount_type IN ('percentage', 'fixed_amount')),
  discount_value INTEGER NOT NULL CHECK (discount_value > 0), -- percentage (1-100) or amount in cents

  -- Usage limits
  usage_limit INTEGER, -- NULL = unlimited total uses
  usage_limit_per_user INTEGER DEFAULT 1, -- NULL = unlimited per user
  usage_count INTEGER NOT NULL DEFAULT 0,

  -- Validity period
  valid_from TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  valid_until TIMESTAMPTZ, -- NULL = no expiration

  -- Conditions
  minimum_order_cents INTEGER NOT NULL DEFAULT 0 CHECK (minimum_order_cents >= 0),
  applicable_categories UUID[] DEFAULT '{}', -- Empty = all categories
  applicable_products UUID[] DEFAULT '{}', -- Empty = all products
  excluded_products UUID[] DEFAULT '{}',

  -- Status
  is_active BOOLEAN NOT NULL DEFAULT TRUE,
  is_public BOOLEAN NOT NULL DEFAULT TRUE, -- If false, only admin can apply

  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Validate coupon and calculate discount
CREATE OR REPLACE FUNCTION validate_coupon(
  p_code TEXT,
  p_user_id UUID DEFAULT NULL,
  p_order_subtotal_cents INTEGER,
  p_category_ids UUID[] DEFAULT '{}',
  p_product_ids UUID[] DEFAULT '{}'
)
RETURNS TABLE (
  valid BOOLEAN,
  coupon_id UUID,
  discount_type TEXT,
  discount_value INTEGER,
  discount_cents INTEGER,
  error_message TEXT
) AS $$
DECLARE
  v_coupon RECORD;
  v_user_usage_count INTEGER := 0;
  v_discount_cents INTEGER := 0;
  v_min_order_ok BOOLEAN := TRUE;
  v_category_ok BOOLEAN := TRUE;
  v_product_ok BOOLEAN := TRUE;
  v_excluded_ok BOOLEAN := TRUE;
BEGIN
  -- Find coupon by code (case insensitive)
  SELECT * INTO v_coupon
  FROM coupons
  WHERE code ILIKE p_code
  LIMIT 1;

  IF NOT FOUND THEN
    RETURN QUERY SELECT FALSE, NULL, NULL, NULL, NULL, 'Coupon invalide';
  END IF;

  -- Check if active
  IF NOT v_coupon.is_active THEN
    RETURN QUERY SELECT FALSE, v_coupon.id, NULL, NULL, NULL, 'Coupon inactif';
  END IF;

  -- Check validity period
  IF v_coupon.valid_from > NOW() THEN
    RETURN QUERY SELECT FALSE, v_coupon.id, NULL, NULL, NULL, 'Coupon pas encore valide';
  END IF;

  IF v_coupon.valid_until IS NOT NULL AND v_coupon.valid_until < NOW() THEN
    RETURN QUERY SELECT FALSE, v_coupon.id, NULL, NULL, NULL, 'Coupon expiré';
  END IF;

  -- Check minimum order
  IF p_order_subtotal_cents < v_coupon.minimum_order_cents THEN
    RETURN QUERY SELECT FALSE, v_coupon.id, NULL, NULL, NULL,
      'Minimum de commande requis: ' || (v_coupon.minimum_order_cents / 100) || ' MAD';
  END IF;

  -- Check total usage limit
  IF v_coupon.usage_limit IS NOT NULL AND v_coupon.usage_count >= v_coupon.usage_limit THEN
    RETURN QUERY SELECT FALSE, v_coupon.id, NULL, NULL, NULL, 'Limite d\'utilisation atteinte';
  END IF;

  -- Check per-user usage limit (if authenticated)
  IF p_user_id IS NOT NULL AND v_coupon.usage_limit_per_user IS NOT NULL THEN
    SELECT COUNT(*) INTO v_user_usage_count
    FROM coupon_usages
    WHERE coupon_id = v_coupon.id AND user_id = p_user_id;

    IF v_user_usage_count >= v_coupon.usage_limit_per_user THEN
      RETURN QUERY SELECT FALSE, v_coupon.id, NULL, NULL, NULL,
        'Vous avez déjà utilisé ce coupon ' || v_coupon.usage_limit_per_user || ' fois';
    END IF;
  END IF;

  -- Check category applicability
  IF array_length(v_coupon.applicable_categories, 1) > 0 THEN
    v_category_ok := FALSE;
    FOR i IN 1..array_length(p_category_ids, 1) LOOP
      IF p_category_ids[i] = ANY(v_coupon.applicable_categories) THEN
        v_category_ok := TRUE;
        EXIT;
      END IF;
    END LOOP;
    IF NOT v_category_ok THEN
      RETURN QUERY SELECT FALSE, v_coupon.id, NULL, NULL, NULL, 'Coupon non applicable aux produits sélectionnés';
    END IF;
  END IF;

  -- Check product applicability
  IF array_length(v_coupon.applicable_products, 1) > 0 THEN
    v_product_ok := FALSE;
    FOR i IN 1..array_length(p_product_ids, 1) LOOP
      IF p_product_ids[i] = ANY(v_coupon.applicable_products) THEN
        v_product_ok := TRUE;
        EXIT;
      END IF;
    END LOOP;
    IF NOT v_product_ok THEN
      RETURN QUERY SELECT FALSE, v_coupon.id, NULL, NULL, NULL, 'Coupon non applicable aux produits sélectionnés';
    END IF;
  END IF;

  -- Check excluded products
  IF array_length(v_coupon.excluded_products, 1) > 0 THEN
    v_excluded_ok := TRUE;
    FOR i IN 1..array_length(p_product_ids, 1) LOOP
      IF p_product_ids[i] = ANY(v_coupon.excluded_products) THEN
        v_excluded_ok := FALSE;
        EXIT;
      END IF;
    END LOOP;
    IF NOT v_excluded_ok THEN
      RETURN QUERY SELECT FALSE, v_coupon.id, NULL, NULL, NULL, 'Coupon non applicable à certains produits';
    END IF;
  END IF;

  -- Calculate discount
  IF v_coupon.discount_type = 'percentage' THEN
    v_discount_cents := (p_order_subtotal_cents * v_coupon.discount_value) / 100;
  ELSE
    v_discount_cents := v_coupon.discount_value;
  END IF;

  -- Cap discount at subtotal
  IF v_discount_cents > p_order_subtotal_cents THEN
    v_discount_cents := p_order_subtotal_cents;
  END IF;

  RETURN QUERY SELECT
    TRUE,
    v_coupon.id,
    v_coupon.discount_type,
    v_coupon.discount_value,
    v_discount_cents,
    NULL;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Check if user can review a product (has delivered order with product)

  can_review BOOLEAN,
  has_delivered_order BOOLEAN,
  existing_review_id UUID
) AS $$
DECLARE
  v_order_exists BOOLEAN;
  v_review_id UUID;
BEGIN
  -- Check if user has a delivered order with this product
  SELECT EXISTS (
    SELECT 1 FROM orders o
    JOIN order_items oi ON oi.order_id = o.id
    WHERE o.user_id = p_user_id

  GET DIAGNOSTICS v_deleted = ROW_COUNT;
  RETURN v_deleted;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- 4. TRIGGERS
-- ============================================================

-- Update updated_at column
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS update_product_reviews_updated_at ON product_reviews;
CREATE TRIGGER update_product_reviews_updated_at
  BEFORE UPDATE ON product_reviews
  FOR EACH ROW
  EXECUTE FUNCTION update_updated_at_column();
